! ============================================================
! CONFIGURACIÓN DEL SWITCH - SW-31
! Proyecto: AEGIS
! ============================================================

enable
configure terminal


! ------------------------------------------------------------
! CONFIGURACIÓN BÁSICA DEL SWITCH
! ------------------------------------------------------------

hostname sw-31
no ip domain-lookup


! ------------------------------------------------------------
! CREACIÓN DE VLANs
! ------------------------------------------------------------

vlan 330
 name THREATINTEL
exit

vlan 340
 name REDTEAM
exit

vlan 350
 name SOC
exit


! ------------------------------------------------------------
! CONFIGURACIÓN DEL ENLACE TRONCAL
! Ethernet0/0 → Uplink hacia SW-5
! ------------------------------------------------------------

interface Ethernet0/0
 description UPLINK HACIA SW-5
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 330,340,350
 no shutdown
exit


! ------------------------------------------------------------
! PUERTO DE ACCESO - THREAT INTELLIGENCE
! Ethernet0/1 → VLAN 330
! ------------------------------------------------------------

interface Ethernet0/1
 description PC Threat Intelligence
 switchport mode access
 switchport access vlan 330
 no shutdown
exit


! ------------------------------------------------------------
! PUERTO DE ACCESO - RED TEAM
! Ethernet0/2 → VLAN 340
! ------------------------------------------------------------

interface Ethernet0/2
 description PC Red Team
 switchport mode access
 switchport access vlan 340
 no shutdown
exit


! ------------------------------------------------------------
! PUERTO DE ACCESO - SOC
! Ethernet0/3 → VLAN 350
! ------------------------------------------------------------

interface Ethernet0/3
 description PC SOC
 switchport mode access
 switchport access vlan 350
 no shutdown
exit


! ------------------------------------------------------------
! SEGURIDAD Y ACCESO ADMINISTRATIVO
! ------------------------------------------------------------

enable secret AEGIS-2026

username admin privilege 15 secret AEGIS-2026


! ------------------------------------------------------------
! CONFIGURACIÓN DE LA CONSOLA
! ------------------------------------------------------------

line con 0
 logging synchronous
 exec-timeout 5 0
exit


! ------------------------------------------------------------
! CONFIGURACIÓN DE ACCESO VTY - SSH
! ------------------------------------------------------------

line vty 0 4
 login local
 transport input ssh
 exec-timeout 5 0
exit


! ------------------------------------------------------------
! FIN DE LA CONFIGURACIÓN
! ------------------------------------------------------------

end
write memory

! FIN DE LA CONFIGURACIÓN DE SW-31
! ============================================================
