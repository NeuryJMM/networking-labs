! ============================================================
! CONFIGURACIÓN DEL SWITCH - SW-30
! Proyecto: AEGIS
! ============================================================

enable
configure terminal

! ------------------------------------------------------------
! CONFIGURACIÓN BÁSICA DEL SWITCH
! ------------------------------------------------------------

hostname sw-30
no ip domain-lookup


! ------------------------------------------------------------
! CREACIÓN DE VLANs
! ------------------------------------------------------------

vlan 310
 name DFIR
exit

vlan 320
 name MALWARE
exit


! ------------------------------------------------------------
! CONFIGURACIÓN DEL ENLACE TRONCAL
! Ethernet0/0 → Uplink hacia SW-5
! ------------------------------------------------------------

interface Ethernet0/0
 description UPLINK HACIA SW-5
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 310,320
 no shutdown
exit


! ------------------------------------------------------------
! PUERTO DE ACCESO - PC DFIR
! Ethernet0/1 → VLAN 310
! ------------------------------------------------------------

interface Ethernet0/1
 description PC DFIR
 switchport mode access
 switchport access vlan 310
 no shutdown
exit


! ------------------------------------------------------------
! PUERTO DE ACCESO - PC MALWARE
! Ethernet0/2 → VLAN 320
! ------------------------------------------------------------

interface Ethernet0/2
 description PC Malware
 switchport mode access
 switchport access vlan 320
 no shutdown
exit


! ------------------------------------------------------------
! SEGURIDAD Y ACCESO ADMINISTRATIVO
! ------------------------------------------------------------

enable secret AEGIS-2026

username admin privilege 15 secret AEGIS-2026


! ------------------------------------------------------------
! CONFIGURACIÓN DE LA CONSOLA
! ------------------------------------------------------------

line con 0
 logging synchronous
 exec-timeout 5 0
exit


! ------------------------------------------------------------
! CONFIGURACIÓN DE ACCESO VTY - SSH
! ------------------------------------------------------------

line vty 0 4
 login local
 transport input ssh
 exec-timeout 5 0
exit


! ------------------------------------------------------------
! GUARDAR CONFIGURACIÓN
! ------------------------------------------------------------

end
write memory

! ============================================================
! FIN DE LA CONFIGURACIÓN DE SW-30
! ============================================================
